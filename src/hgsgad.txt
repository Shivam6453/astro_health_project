// src/components/HealthOptions.js
import React, { useState } from "react";
import { useNavigate } from "react-router-dom";

export default function HealthOptions() {
  const navigate = useNavigate();
  
  const [savedSections, setSavedSections] = useState({
    physicalHealth: false,
    mentalHealth: false,
    medications: false,
    healthRecords: false,
  });

  const [healthData, setHealthData] = useState({
    physicalHealth: { heartRate: "", bloodPressure: "", temperature: "" },
    mentalHealth: { mood: "5", stress: "5", sleep: "7" },
    medications: { currentMeds: "", dosage: "", sideEffects: "" },
    healthRecords: { pastIllnesses: "", surgeries: "", vaccinations: "" },
  });

  const [activeSection, setActiveSection] = useState(null);
  const [errors, setErrors] = useState({});

  const validateSection = (sectionName) => {
    const section = healthData[sectionName];
    const newErrors = {};

    if (sectionName === "physicalHealth") {
      if (!section.heartRate.trim()) newErrors.heartRate = "Heart rate required";
      if (!section.bloodPressure.trim()) newErrors.bloodPressure = "Blood pressure required";
      if (!section.temperature.trim()) newErrors.temperature = "Temperature required";
    } else if (sectionName === "medications") {
      if (!section.currentMeds.trim()) newErrors.currentMeds = "Medications required";
      if (!section.dosage.trim()) newErrors.dosage = "Dosage required";
    } else if (sectionName === "healthRecords") {
      if (!section.pastIllnesses.trim()) newErrors.pastIllnesses = "Past illnesses required";
      if (!section.vaccinations.trim()) newErrors.vaccinations = "Vaccinations required";
    }

    return newErrors;
  };

  const handleSaveSection = (sectionName) => {
    const sectionErrors = validateSection(sectionName);
    
    if (Object.keys(sectionErrors).length > 0) {
      setErrors(sectionErrors);
      alert(`⚠️ Please fill all fields in ${sectionName}`);
      return;
    }

    setSavedSections(prev => ({
      ...prev,
      [sectionName]: true
    }));
    setErrors({});
    alert(`✓ ${sectionName} saved successfully!`);
  };

  const handleInputChange = (sectionName, fieldName, value) => {
    setHealthData(prev => ({
      ...prev,
      [sectionName]: {
        ...prev[sectionName],
        [fieldName]: value
      }
    }));
    if (errors[fieldName]) {
      setErrors(prev => ({
        ...prev,
        [fieldName]: ""
      }));
    }
  };

  const canProceed = Object.values(savedSections).every(val => val === true);

  const SectionCard = ({ sectionName, sectionTitle, icon, children }) => {
    const isSaved = savedSections[sectionName];

    return (
      <div style={{
        background: isSaved 
          ? "linear-gradient(135deg, rgba(0, 255, 165, 0.1), rgba(0, 255, 100, 0.1))"
          : "linear-gradient(135deg, rgba(102, 126, 234, 0.1), rgba(183, 0, 255, 0.1))",
        border: isSaved 
          ? "2px solid rgba(0, 255, 165, 0.5)"
          : "2px solid rgba(102, 126, 234, 0.3)",
        borderRadius: "10px",
        padding: "1.5rem",
        marginBottom: "1.5rem",
        transition: "all 0.3s"
      }}>
        <div style={{
          display: "flex",
          justifyContent: "space-between",
          alignItems: "center",
          marginBottom: "1rem"
        }}>
          <h3 style={{
            color: isSaved ? "#00FFA5" : "#00FFFF",
            fontWeight: "700",
            fontSize: "1.2rem"
          }}>
            {icon} {sectionTitle}
          </h3>
          {isSaved && (
            <span style={{
              background: "#00FFA5",
              color: "#000",
              padding: "5px 10px",
              borderRadius: "20px",
              fontWeight: "700",
              fontSize: "0.85rem"
            }}>
              ✓ Saved
            </span>
          )}
        </div>

        {!isSaved ? (
          <>
            <div onClick={() => setActiveSection(activeSection === sectionName ? null : sectionName)}
              style={{
                cursor: "pointer",
                padding: "10px",
                background: "rgba(255, 255, 255, 0.05)",
                borderRadius: "6px",
                marginBottom: "1rem",
                color: "#667eea",
                fontWeight: "600"
              }}>
              {activeSection === sectionName ? "▼ Hide Details" : "▶ Show Details"}
            </div>

            {activeSection === sectionName && (
              <>
                {children}
                <button
                  onClick={() => handleSaveSection(sectionName)}
                  style={{
                    width: "100%",
                    padding: "12px",
                    background: "linear-gradient(135deg, #00FFFF, #667eea)",
                    border: "none",
                    borderRadius: "6px",
                    color: "#000",
                    fontWeight: "700",
                    cursor: "pointer",
                    marginTop: "1rem",
                    transition: "all 0.3s"
                  }}
                  onMouseOver={(e) => (e.target.style.boxShadow = "0 0 20px rgba(0, 255, 255, 0.5)")}
                  onMouseOut={(e) => (e.target.style.boxShadow = "none")}
                >
                  💾 Save {sectionTitle}
                </button>
              </>
            )}
          </>
        ) : (
          <p style={{ color: "#888" }}>
            {sectionName === "physicalHealth" && `❤️ Heart Rate: ${healthData.physicalHealth.heartRate} | BP: ${healthData.physicalHealth.bloodPressure}`}
            {sectionName === "mentalHealth" && `🧠 Mood: ${healthData.mentalHealth.mood}/10 | Stress: ${healthData.mentalHealth.stress}/10`}
            {sectionName === "medications" && `💊 Medications saved`}
            {sectionName === "healthRecords" && `📋 Health records saved`}
          </p>
        )}
      </div>
    );
  };

  const InputField = ({ label, name, sectionName, type = "text", placeholder }) => (
    <div style={{ marginBottom: "1rem" }}>
      <label style={{
        display: "block",
        marginBottom: "5px",
        fontWeight: "600",
        color: errors[name] ? "#ff6b6b" : "#e0e0e0",
        fontSize: "0.9rem"
      }}>
        {label} <span style={{ color: "#ff6b6b" }}>*</span>
      </label>
      <input
        type={type}
        placeholder={placeholder}
        value={healthData[sectionName][name]}
        onChange={(e) => handleInputChange(sectionName, name, e.target.value)}
        style={{
          width: "100%",
          padding: "10px 12px",
          border: errors[name] ? "2px solid #ff6b6b" : "2px solid rgba(102, 126, 234, 0.3)",
          borderRadius: "6px",
          background: "rgba(255, 255, 255, 0.05)",
          color: "#ffffff",
          fontSize: "0.95rem",
          boxSizing: "border-box",
          transition: "all 0.3s",
          caretColor: "#667eea"
        }}
        onFocus={(e) => {
          if (!errors[name]) {
            e.target.style.borderColor = "#667eea";
            e.target.style.boxShadow = "0 0 10px rgba(102, 126, 234, 0.3)";
          }
        }}
      />
      {errors[name] && <p style={{ color: "#ff6b6b", fontSize: "0.8rem", marginTop: "3px" }}>⚠ {errors[name]}</p>}
    </div>
  );

  const TextAreaField = ({ label, name, sectionName, placeholder, rows = 3 }) => (
    <div style={{ marginBottom: "1rem" }}>
      <label style={{
        display: "block",
        marginBottom: "5px",
        fontWeight: "600",
        color: errors[name] ? "#ff6b6b" : "#e0e0e0",
        fontSize: "0.9rem"
      }}>
        {label} <span style={{ color: "#ff6b6b" }}>*</span>
      </label>
      <textarea
        placeholder={placeholder}
        rows={rows}
        value={healthData[sectionName][name]}
        onChange={(e) => handleInputChange(sectionName, name, e.target.value)}
        style={{
          width: "100%",
          padding: "10px 12px",
          border: errors[name] ? "2px solid #ff6b6b" : "2px solid rgba(102, 126, 234, 0.3)",
          borderRadius: "6px",
          background: "rgba(255, 255, 255, 0.05)",
          color: "#ffffff",
          fontSize: "0.95rem",
          boxSizing: "border-box",
          fontFamily: "inherit",
          resize: "vertical",
          transition: "all 0.3s",
          caretColor: "#667eea"
        }}
        onFocus={(e) => {
          if (!errors[name]) {
            e.target.style.borderColor = "#667eea";
            e.target.style.boxShadow = "0 0 10px rgba(102, 126, 234, 0.3)";
          }
        }}
      />
      {errors[name] && <p style={{ color: "#ff6b6b", fontSize: "0.8rem", marginTop: "3px" }}>⚠ {errors[name]}</p>}
    </div>
  );

  const SliderField = ({ label, name, sectionName, min = 1, max = 10 }) => (
    <div style={{ marginBottom: "1rem" }}>
      <label style={{
        display: "block",
        marginBottom: "5px",
        fontWeight: "600",
        color: "#e0e0e0",
        fontSize: "0.9rem"
      }}>
        {label}: <span style={{ color: "#667eea", fontWeight: "700" }}>{healthData[sectionName][name]}/10</span>
      </label>
      <input
        type="range"
        min={min}
        max={max}
        value={healthData[sectionName][name]}
        onChange={(e) => handleInputChange(sectionName, name, e.target.value)}
        style={{
          width: "100%",
          cursor: "pointer"
        }}
      />
    </div>
  );

  return (
    <div style={{
      minHeight: "100vh",
      background: "linear-gradient(135deg, #0f0f23 0%, #1a0a3e 100%)",
      padding: "40px 20px",
      fontFamily: "'Segoe UI', Tahoma, Geneva, Verdana, sans-serif"
    }}>
      <section style={{
        maxWidth: "900px",
        margin: "0 auto",
        background: "linear-gradient(135deg, #1a1a2e 0%, #16213e 100%)",
        borderRadius: "15px",
        padding: "40px",
        boxShadow: "0 20px 60px rgba(0, 0, 0, 0.5)",
        border: "2px solid rgba(102, 126, 234, 0.2)"
      }}>
        <div style={{ marginBottom: "2rem", textAlign: "center" }}>
          <h1 style={{
            fontSize: "2rem",
            fontWeight: "700",
            background: "linear-gradient(135deg, #00FFFF, #B700FF)",
            WebkitBackgroundClip: "text",
            WebkitTextFillColor: "transparent",
            marginBottom: "0.5rem"
          }}>
            🏥 Health Sections
          </h1>
          <p style={{ color: "#888", fontSize: "0.95rem" }}>
            Complete all sections and save before proceeding
          </p>
          <div style={{
            marginTop: "1rem",
            display: "flex",
            gap: "10px",
            justifyContent: "center",
            flexWrap: "wrap"
          }}>
            {Object.entries(savedSections).map(([key, value]) => (
              <span key={key} style={{
                padding: "5px 15px",
                background: value ? "rgba(0, 255, 165, 0.2)" : "rgba(255, 107, 107, 0.2)",
                border: `2px solid ${value ? "#00FFA5" : "#ff6b6b"}`,
                borderRadius: "20px",
                color: value ? "#00FFA5" : "#ff6b6b",
                fontWeight: "600",
                fontSize: "0.85rem"
              }}>
                {value ? "✓" : "○"} {key}
              </span>
            ))}
          </div>
        </div>

        {/* Physical Health */}
        <SectionCard sectionName="physicalHealth" sectionTitle="Physical Health" icon="❤️">
          <InputField label="Heart Rate (BPM)" name="heartRate" sectionName="physicalHealth" type="number" placeholder="e.g., 72" />
          <InputField label="Blood Pressure (mmHg)" name="bloodPressure" sectionName="physicalHealth" placeholder="e.g., 120/80" />
          <InputField label="Temperature (°C)" name="temperature" sectionName="physicalHealth" type="number" placeholder="e.g., 37.2" />
        </SectionCard>

        {/* Mental Health */}
        <SectionCard sectionName="mentalHealth" sectionTitle="Mental Health" icon="🧠">
          <SliderField label="Mood" name="mood" sectionName="mentalHealth" />
          <SliderField label="Stress Level" name="stress" sectionName="mentalHealth" />
          <SliderField label="Sleep Quality" name="sleep" sectionName="mentalHealth" />
        </SectionCard>

        {/* Medications */}
        <SectionCard sectionName="medications" sectionTitle="Medications" icon="💊">
                    <InputField label="Current Medications" name="currentMeds" sectionName="medications" placeholder="List all medications" />
          <InputField label="Dosage" name="dosage" sectionName="medications" placeholder="e.g., 500mg twice daily" />
          <TextAreaField label="Side Effects (if any)" name="sideEffects" sectionName="medications" placeholder="List any side effects" rows={2} />
        </SectionCard>

        {/* Health Records */}
        <SectionCard sectionName="healthRecords" sectionTitle="Health Records" icon="📋">
          <TextAreaField label="Past Illnesses" name="pastIllnesses" sectionName="healthRecords" placeholder="List previous illnesses" rows={2} />
          <InputField label="Surgeries" name="surgeries" sectionName="healthRecords" placeholder="List any surgeries" />
          <InputField label="Vaccinations" name="vaccinations" sectionName="healthRecords" placeholder="List vaccinations received" />
        </SectionCard>

        {/* Action Buttons */}
        <div style={{
          display: "grid",
          gridTemplateColumns: "1fr 1fr",
          gap: "1rem",
          marginTop: "2rem"
        }}>
          <button
            onClick={() => navigate("/")}
            style={{
              padding: "12px 20px",
              background: "rgba(255, 100, 0, 0.2)",
              border: "2px solid #FF6400",
              borderRadius: "6px",
              color: "#FF6400",
              fontWeight: "700",
              fontSize: "14px",
              cursor: "pointer",
              transition: "all 0.3s",
            }}
            onMouseOver={(e) => (e.target.style.background = "rgba(255, 100, 0, 0.4)")}
            onMouseOut={(e) => (e.target.style.background = "rgba(255, 100, 0, 0.2)")}
          >
            ← Back to Home
          </button>

          <button
            onClick={() => {
              if (canProceed) {
                sessionStorage.setItem("healthSectionsData", JSON.stringify(healthData));
                navigate("/dashboard");
              } else {
                alert("⚠️ Please complete and save all sections first!");
              }
            }}
            disabled={!canProceed}
            style={{
              padding: "12px 20px",
              background: canProceed 
                ? "linear-gradient(135deg, #00FFA5, #00FF88)"
                : "linear-gradient(135deg, rgba(0, 255, 165, 0.3), rgba(0, 255, 136, 0.3))",
              border: "none",
              borderRadius: "6px",
              color: "#000",
              fontWeight: "700",
              fontSize: "14px",
              cursor: canProceed ? "pointer" : "not-allowed",
              transition: "all 0.3s",
              opacity: canProceed ? 1 : 0.6,
              boxShadow: canProceed ? "0 0 20px rgba(0, 255, 165, 0.3)" : "none"
            }}
            onMouseOver={(e) => {
              if (canProceed) {
                e.target.style.boxShadow = "0 0 30px rgba(0, 255, 165, 0.6)";
              }
            }}
            onMouseOut={(e) => {
              if (canProceed) {
                e.target.style.boxShadow = "0 0 20px rgba(0, 255, 165, 0.3)";
              }
            }}
          >
            {canProceed ? "✓ All Sections Complete - Go to Dashboard" : "⊘ Complete All Sections"}
          </button>
        </div>

        {canProceed && (
          <div style={{
            background: "linear-gradient(135deg, rgba(0, 255, 165, 0.15), rgba(0, 255, 100, 0.15))",
            border: "2px solid #00FFA5",
            borderRadius: "10px",
            padding: "1rem",
            textAlign: "center",
            color: "#00FFA5",
            fontWeight: "700",
            fontSize: "14px",
            marginTop: "1rem"
          }}>
            ✓ All health sections completed! Ready to proceed to dashboard.
          </div>
        )}
      </section>
    </div>
  );
}
