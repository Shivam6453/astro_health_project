import React, { useState } from "react";

export default function AstronautDetails() {
  const [formData, setFormData] = useState({
    fullName: "",
    age: "",
    height: "",
    weight: "",
    bloodType: "",
    medicalHistory: "",
    allergies: "",
    medications: "",
    fitnessLevel: "",
    spaceMissions: "",
    emergencyContact: "",
    notes: "",
  });

  const [submitted, setSubmitted] = useState(false);

  // Handle input changes
  function handleInputChange(e) {
    const { name, value } = e.target;
    setFormData((prev) => ({
      ...prev,
      [name]: value,
    }));
  }

  // Export to text file
  function handleExportToFile() {
    // Check if any field is filled
    const hasData = Object.values(formData).some((value) => value.trim() !== "");

    if (!hasData) {
      alert("Please fill at least one field before exporting!");
      return;
    }

    // Create text content
    const textContent = `
╔════════════════════════════════════════════════════════════════╗
║            ASTRONAUT HEALTH PROFILE - CONFIDENTIAL             ║
║                  Generated: ${new Date().toLocaleString()}          ║
╚════════════════════════════════════════════════════════════════╝

═══ PERSONAL INFORMATION ═══
Full Name:              ${formData.fullName || "N/A"}
Age:                    ${formData.age || "N/A"} years
Height:                 ${formData.height || "N/A"} cm
Weight:                 ${formData.weight || "N/A"} kg
Blood Type:             ${formData.bloodType || "N/A"}

═══ MEDICAL INFORMATION ═══
Medical History:        ${formData.medicalHistory || "N/A"}
Allergies:              ${formData.allergies || "N/A"}
Current Medications:    ${formData.medications || "N/A"}
Fitness Level:          ${formData.fitnessLevel || "N/A"}

═══ MISSION INFORMATION ═══
Previous Space Missions: ${formData.spaceMissions || "N/A"}

═══ EMERGENCY CONTACT ═══
Contact Details:        ${formData.emergencyContact || "N/A"}

═══ ADDITIONAL NOTES ═══
${formData.notes || "N/A"}

────────────────────────────────────────────────────────────────
File generated by Astro_Friend Health Platform
This file contains sensitive personal health information.
Handle with care and store securely.
────────────────────────────────────────────────────────────────
    `;

    // Create blob and download
    const blob = new Blob([textContent], { type: "text/plain" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = `astronaut_profile_${formData.fullName || "unknown"}_${Date.now()}.txt`;
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    URL.revokeObjectURL(url);

    alert("✓ Profile exported successfully!");
  }

  // Clear form
  function handleClearForm() {
    if (window.confirm("Are you sure you want to clear all fields?")) {
      setFormData({
        fullName: "",
        age: "",
        height: "",
        weight: "",
        bloodType: "",
        medicalHistory: "",
        allergies: "",
        medications: "",
        fitnessLevel: "",
        spaceMissions: "",
        emergencyContact: "",
        notes: "",
      });
      setSubmitted(false);
    }
  }

  function handleSubmit(e) {
    e.preventDefault();
    setSubmitted(true);
  }

          }}>

              📋 Personal Information

              <div>
                <label style={{ display: "block", marginBottom: "6px", fontWeight: "600", color: "#00FFFF" }}>
                  Full Name *
                </label>
                <input
                  type="text"

                  value={formData.fullName}

                    color: "#00FFFF",
                    fontSize: "14px",
                    boxSizing: "border-box",
                  }}
                />
              </div>

              <div>
                <label style={{ display: "block", marginBottom: "6px", fontWeight: "600", color: "#00FFFF" }}>
                  Weight (kg)
                </label>
                <input
                  type="number"
                  name="weight"
                  value={formData.weight}
                  onChange={handleInputChange}
                  placeholder="Weight in kilograms"
                  style={{
                    width: "100%",
                    padding: "10px 12px",
                    border: "2px solid #00FFFF",
                    borderRadius: "6px",
                    background: "rgba(0, 255, 255, 0.05)",
                    color: "#00FFFF",

                    boxSizing: "border-box",
                  }}
                />
              </div>

              <div>

                  Blood Type

                <select
                  name="bloodType"
                  value={formData.bloodType}

                <label style={{ display: "block", marginBottom: "6px", fontWeight: "600", color: "#FF006E" }}>
                  Medical History
                </label>
                <textarea
                  name="medicalHistory"
                  value={formData.medicalHistory}
                  onChange={handleInputChange}
                  placeholder="Previous medical conditions, surgeries, etc."

                  style={{
                    width: "100%",
                    padding: "10px 12px",
                    border: "2px solid #FF006E",
                    borderRadius: "6px",
                    background: "rgba(255, 0, 110, 0.05)",
                    color: "#FF006E",
                    fontSize: "14px",
                    boxSizing: "border-box",
                    fontFamily: "inherit",
                  }}

              </div>

                <label style={{ display: "block", marginBottom: "6px", fontWeight: "600", color: "#FF006E" }}>

                    borderRadius: "6px",

                  <option value="Good">Good</option>

                              <label style={{ display: "block", marginBottom: "6px", fontWeight: "600", color: "#00FFA5" }}>

            border: "2px solid #FF6400",
            borderRadius: "10px",

            <h2 style={{ color: "#FF6400", marginBottom: "1rem", fontWeight: "700" }}>

                Emergency Contact Details

              <textarea
                name="emergencyContact"
